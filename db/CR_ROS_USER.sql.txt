CREATE USER ROS
  IDENTIFIED BY &ros_password
  DEFAULT TABLESPACE ROS_DATA
  TEMPORARY TABLESPACE TEMP
  PROFILE DEFAULT
  ACCOUNT UNLOCK
/
  GRANT RESOURCE TO ROS
/
  GRANT CONNECT TO ROS
/
  ALTER USER ROS DEFAULT ROLE ALL
/
  ALTER USER ROS QUOTA UNLIMITED ON ROS_DATA
/
  ALTER USER ROS QUOTA UNLIMITED ON ROS_IDX
/
